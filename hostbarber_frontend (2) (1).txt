function renderTicket(){
  const itemsEl = document.getElementById('ticketItems');
  const emptyEl = document.getElementById('ticketEmpty');
  itemsEl.innerHTML = '';
  emptyEl.style.display = ticket.length ? 'none' : 'block';
  let subtotal = 0;
  ticket.forEach((it,i)=>{
    subtotal += it.price;
    const row = document.createElement('div');
    row.className = 'ticket-line';
    row.innerHTML = `<span>${it.name}</span><span class="mono">${fmt(it.price)}<span class="remove" onclick="removeTicketItem(${i})">remover</span></span>`;
    itemsEl.appendChild(row);
  });
  const discountPct = parseFloat(document.getElementById('discountInput').value) || 0;
  const tip = parseFloat(document.getElementById('tipInput').value) || 0;
  const discountVal = subtotal * (discountPct/100);
  const activePay = document.querySelector('.pay-tab.active').dataset.p;
  let total = Math.max(0, subtotal - discountVal + tip);
  if(activePay === 'assinatura') total = tip; // serviço coberto pelo plano, só gorjeta é cobrada

  document.getElementById('ttSubtotal').textContent = fmt(subtotal);
  document.getElementById('ttTotal').textContent = fmt(total);

  // troco
  const received = parseFloat(document.getElementById('receivedInput').value) || 0;
  const troco = received - total;
  const trocoDisplay = document.getElementById('trocoDisplay');
  trocoDisplay.classList.toggle('negative', troco < 0);
  trocoDisplay.innerHTML = troco < 0
    ? `Falta: <span class="mono">${fmt(Math.abs(troco))}</span>`
    : `Troco: <span class="mono">${fmt(troco)}</span>`;

  document.getElementById('ticketClientLabel').textContent = document.getElementById('clientSelect').value;
}

function renderSplit(){
  const total = parseFloat(document.getElementById('ttTotal').textContent.replace('R$','').replace('.','').replace(',','.')) || 0;
  const a = parseFloat(document.getElementById('splitAmountA').value) || 0;
  const b = parseFloat(document.getElementById('splitAmountB').value) || 0;
  const remaining = total - a - b;
  const el = document.getElementById('splitRemaining');
  el.classList.toggle('ok', Math.abs(remaining) < 0.01);
  el.textContent = Math.abs(remaining) < 0.01 ? 'Valores conferem ✓' : ('Falta dividir: ' + fmt(Math.max(0,remaining)));
}

function finalizeTicket(){
  if(ticket.length === 0){ alert('Adicione ao menos um serviço à comanda.'); return; }
  const client = document.getElementById('clientSelect').value;
  const pro = document.getElementById('proSelect').value;
  const activePay = document.querySelector('.pay-tab.active').dataset.p;
  const totalText = document.getElementById('ttTotal').textContent;
  const isSplit = document.getElementById('splitToggle').checked;
  const payLabel = isSplit ? 'Dividido' : {dinheiro:'Dinheiro', debito:'Débito', credito:'Crédito', pix:'Pix', assinatura:'Assinatura'}[activePay];

  const row = document.createElement('div');
  row.className = 'sh-row';
  const now = new Date();
  const hh = String(now.getHours()).padStart(2,'0')+':'+String(now.getMinutes()).padStart(2,'0');
  row.innerHTML = `<div class="sh-left"><span class="sh-name">${client}</span><span class="sh-meta">${hh} · ${pro} · ${payLabel}</span></div><span class="sh-val">${totalText}</span>`;
  document.getElementById('salesHistory').prepend(row);

  // build receipt
  const receiptBody = document.getElementById('receiptBody');
  receiptBody.innerHTML = ticket.map(it=>`<div class="receipt-line"><span>${it.name}</span><span class="mono">${fmt(it.price)}</span></div>`).join('')
    + `<div class="receipt-line"><span>Cliente</span><span>${client}</span></div>`
    + `<div class="receipt-line"><span>Profissional</span><span>${pro}</span></div>`
    + `<div class="receipt-line"><span>Forma de pagamento</span><span>${payLabel}</span></div>`
    + `<div class="receipt-line"><span>Horário</span><span>${hh}</span></div>`;
  document.getElementById('receiptTotal').textContent = totalText;
  document.getElementById('receiptOverlay').classList.add('active');

  // reset ticket
  ticket = [];
  document.getElementById('discountInput').value = 0;
  document.getElementById('tipInput').value = 0;
  document.getElementById('receivedInput').value = '';
  document.getElementById('splitToggle').checked = false;
  toggleSplit();
  renderTicket();
}

const stockData = [
  {name:'Pomada matte', qty:24, min:10, img:null},
  {name:'Óleo para barba', qty:6, min:10, img:null},
  {name:'Shampoo carvão', qty:18, min:8, img:null},
  {name:'Pente de madeira', qty:31, min:10, img:null},
  {name:'Navalhas descartáveis (cx)', qty:2, min:5, img:null},
  {name:'Toalhas descartáveis (pct)', qty:14, min:6, img:null},
];
const stockBody = document.getElementById('stockTableBody');
function renderStock(){
  stockBody.innerHTML = '';
  stockData.forEach((s,i)=>{
    const pct = Math.max(4, Math.min(100, Math.round((s.qty/(s.min*2))*100)));
    const status = s.qty < s.min ? 'Crítico' : pct < 50 ? 'Baixo' : 'OK';
    const color = status==='Crítico' ? 'var(--danger)' : status==='Baixo' ? 'var(--brass)' : 'var(--success)';
    const tr = document.createElement('tr');
    if(s.editing){
      tr.innerHTML = `
        <td>${photoCellHtml('stock', i, s.img)}</td>
        <td><input class="edit-input" id="stkName${i}" value="${s.name}"></td>
        <td><input class="edit-input" id="stkQty${i}" type="number" value="${s.qty}"></td>
        <td><input class="edit-input" id="stkMin${i}" type="number" value="${s.min}"></td>
        <td colspan="1" style="color:var(--text-faint);font-size:11px;">nível recalcula ao salvar</td>
        <td><button class="icon-btn" onclick="saveStock(${i})" title="Salvar">✓</button></td>`;
    } else {
      tr.innerHTML = `
        <td>${photoCellHtml('stock', i, s.img)}</td>
        <td>${s.name}</td><td class="mono">${s.qty} un.</td><td class="mono">${s.min} un.</td>
        <td><div class="stock-bar"><div class="stock-bar-fill" style="width:${pct}%;background:${color}"></div></div></td>
        <td><span class="svc-tag" style="color:${color}">${status}</span></td>
        <td style="display:flex;gap:5px;"><button class="icon-btn" onclick="editStock(${i})" title="Editar">✎</button><button class="icon-btn" onclick="quickAddStock(${i})" title="Registrar entrada">+</button></td>`;
    }
    stockBody.appendChild(tr);
  });
}
function editStock(i){ stockData[i].editing = true; renderStock(); }
function saveStock(i){
  stockData[i].name = document.getElementById('stkName'+i).value;
  stockData[i].qty = parseInt(document.getElementById('stkQty'+i).value) || 0;
  stockData[i].min = parseInt(document.getElementById('stkMin'+i).value) || 1;
  stockData[i].editing = false;
  renderStock();
}
function quickAddStock(i){
  const val = prompt('Quantas unidades entraram no estoque de "'+stockData[i].name+'"?');
  if(val && !isNaN(parseInt(val))){ stockData[i].qty += parseInt(val); renderStock(); }
}
function addStockItem(){
  stockData.push({name:'Novo produto', qty:0, min:5, img:null, editing:true});
  renderStock();
}
renderStock();
